cmake_minimum_required(VERSION 3.16)
project(umbra_player_host LANGUAGES CXX)
add_executable(umbra-player-host main.cpp)
target_compile_features(umbra-player-host PRIVATE cxx_std_17)
if(WIN32)
  target_compile_definitions(umbra-player-host PRIVATE _WIN32_WINNT=0x0A00)
  target_link_libraries(umbra-player-host PRIVATE user32 gdi32)
  set_property(TARGET umbra-player-host PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
else()
  find_package(X11 REQUIRED)
  find_package(Threads REQUIRED)
  target_link_libraries(umbra-player-host PRIVATE X11::X11 Threads::Threads)
endif()
